\chapter*{Frobenius Separation of Massive and Massless Sectors in $GF(27)^*$}
\addcontentsline{toc}{chapter}{Frobenius Separation of Massive and Massless Sectors}

\begin{abstract}
The multiplicative group $GF(27)^*\cong\mathbb{Z}_{26}$ decomposes under the
Frobenius automorphism $\phi:x\mapsto x^3$ into exactly two types: two fixed
points $\{+1,-1\}$ and eight three-element orbits. Via the decomposition
$\mathbb{Z}_{26}\cong\mathbb{Z}_2\times\mathbb{Z}_{13}$ (CRT), four
$\mathbb{Z}_{13}$ three-orbits times two $\mathbb{Z}_2$ parities yield the
eight gluons of the adjoint $SU(3)$ representation; the fixed field $GF(3)$
of the Frobenius supplies the $U(1)$ sector (photon). The fixed points
$\{+1,-1\}$ correspond to the massive $\mathbb{Z}_2$ sector
(particle/antiparticle). All results are algebraically forced. Status: \KM.
\end{abstract}

\section*{Decomposition of $\mathbb{Z}_{26}$ under the Frobenius}

\subsection*{Basic structure \BM}

$GF(27)^*\cong\mathbb{Z}_{26}$, cyclic of order $3^3-1=26=2\cdot13$. Since
$\gcd(2,13)=1$:
\[
\mathbb{Z}_{26}\cong\mathbb{Z}_2\times\mathbb{Z}_{13}.
\]
The Frobenius $\phi:x\mapsto x^3$ acts:
\begin{itemize}
\item On $\mathbb{Z}_2$: $3\equiv1\pmod{2}$ — $\phi$ is \emph{trivial}.
\item On $\mathbb{Z}_{13}$: $\mathrm{ord}(3,\mathbb{Z}_{13})=3$ (since
  $3^3=27\equiv1\pmod{13}$) — $\phi$ has order~3.
\end{itemize}

\section*{Physical Identification \KM}

\subsection*{Massive sector: fixed points $=\mathbb{Z}_2$}

The two fixed points $\{+1,-1\}$ form the massive sector: every massive particle
has an antiparticle ($\mathbb{Z}_2$ structure), consistent with the compact
torus sector.

\subsection*{Massless $SU(3)$ sector: eight gluons}

\[
8\ \text{gluons}
=4\ \mathbb{Z}_{13}\text{-orbits}\times2\ \mathbb{Z}_2\text{-parities}
=|GF(9)^*|.
\]
The three elements of each $\mathbb{Z}_{13}$ orbit correspond to the three
colours $\{\mathrm{R},\mathrm{G},\mathrm{B}\}$; the $\mathbb{Z}_2$ parity
distinguishes colour from anticolour. This is the adjoint representation of
$SU(3)$, derived from Galois structure alone.

\subsection*{$U(1)$ sector: photon from the fixed field $GF(3)$}

The fixed field of the Frobenius on $GF(27)$ is $GF(3)$ (Fermat: $x^3=x$ for
$x\in GF(3)$). The projection $GF(27)\to GF(3)$ gives:

\begin{center}
\begin{tabular}{ll}
\toprule
Structure & Sector \\
\midrule
$GF(3)^*=\{1,2\}\cong\mathbb{Z}_2$ & Photon: $U(1)$, own antiparticle \\
$27-3=24$ remaining elements & $SU(3)$ sector (quarks, gluons) \\
\bottomrule
\end{tabular}
\end{center}

\section*{Algebraic Necessity \BM}

\begin{align}
\text{Non-fixed elements in }GF(27)^* &= 26-2=24,\\
24 &= 8\cdot3=|GF(9)^*|\cdot|\text{colours}|
     =(3^2-1)\cdot3=3^3-3,\\
8\ \text{gluons} &= \tfrac{26-2}{3}=8.
\end{align}
All three relations follow solely from $|GF(27)|=3^3$ and the order~3 of the
Frobenius on $\mathbb{Z}_{13}$.
