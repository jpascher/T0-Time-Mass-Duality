\section*{Summary}
% =========================================================

Doc.~321 derives $\sin^2\theta_W|_{\rm GUT} = 3/8$ algebraically
from the trace formula of the five orbifold states~\textbf{[B]}.
The open bridge to the measured value $\sin^2\theta_W(M_Z) = 0.2312$
(PDG) is closed here.

The complete FFGFT formula is:
\begin{equation}
  \boxed{\sin^2\theta_W(M_Z)
  = \frac{3}{8}
  - \frac{55\,\alpha_{\rm em}(M_Z)}{24\pi}
    \Bigl[\ln\!\tfrac{m_{\rm Pl}}{M_Z}
         + \tfrac{19}{12}\ln\xi\Bigr]
  = 0.2308,}
  \label{eq:main}
\end{equation}
at $-0.19\,\%$ deviation from PDG. All inputs come from the FFGFT
corpus. The exponent $p = 19/12 = 3/2 + 1/(4N_c)$ connects the
electron mass exponent ($p_e = 3/2$, Doc.~320) with the
$D_4$-lattice phase correction.

\section{Starting Point and Strategy}
% =========================================================

\textbf{[B]} Doc.~321 derives $\sin^2\theta_W|_{\rm GUT} = 3/8$ from
the trace formula at the unification point. The measured value at
$M_Z$ differs by
\begin{equation}
  \Delta\sin^2\theta_W = 3/8 - 0.2312 = 0.1438,
\end{equation}
which is the RG running between $M_{\rm GUT}$ and $M_Z$.
The derivation proceeds in three steps:
\begin{enumerate}
  \item $\alpha_s(M_Z)$ from the 1-loop RG run of $\alpha_s(m_\tau)$
    -- fully from FFGFT.
  \item GUT scale $M_{\rm GUT} = m_{\rm Pl}\cdot\xi^{19/12}$
    from FFGFT geometry.
  \item $\sin^2\theta_W(M_Z)$ from the Georgi-Quinn-Weinberg formula.
\end{enumerate}

% =========================================================
\section{Step 1: $\alpha_s(M_Z)$ from FFGFT Running}
% =========================================================

\textbf{[B]} With $N_c=3$ (Doc.~321) and $n_f=5$ active quark
flavours between $m_\tau$ and $M_Z$:
\begin{equation}
  \beta_3 = \frac{11N_c - 2n_f}{3} = \frac{23}{3}.
\end{equation}

\textbf{[K]} Integrating the 1-loop RGE:
\begin{equation}
  \frac{1}{\alpha_s(M_Z)}
  = \frac{1}{\alpha_s(m_\tau)} + \frac{\beta_3}{2\pi}\ln\!\frac{M_Z}{m_\tau}.
\end{equation}

Inserting FFGFT values ($\alpha_s(m_\tau) = 3\xi^{1/4} = 0.3224$,
$m_\tau = (25/9)\xi^{2/3}v = 1.783$~GeV, $M_Z = 91.19$~GeV):
\begin{equation}
  \frac{1}{\alpha_s(M_Z)}
  = 3.102 + \frac{23/3}{2\pi}\times3.934
  = 3.102 + 4.800 = 7.902.
\end{equation}
\begin{equation}
  \boxed{\alpha_s(M_Z) = 0.1265}
  \quad(\text{PDG: }0.1181,\;+7.1\,\%).
\end{equation}

The $+7.1\,\%$ discrepancy is the known 1-loop error; 2-loop
corrections and quark-threshold effects (particularly the $b$-quark
threshold) bring it to $\approx0.118$. The Weinberg-angle derivation
depends on $\alpha_s(M_Z)$ only indirectly via $M_{\rm GUT}$ and is
much less sensitive to this error.

% =========================================================
\section{Step 2: The GUT Scale from FFGFT Geometry}
% =========================================================

\textbf{[K]} The GUT scale in FFGFT:
\begin{equation}
  M_{\rm GUT} = m_{\rm Pl}\cdot\xi^{p},
  \quad p = \frac{19}{12} = \frac{3}{2} + \frac{1}{4N_c}.
  \label{eq:MGUT}
\end{equation}

The exponent $p = 19/12$ has a geometric derivation:
\begin{itemize}
  \item $p_e = 3/2$: electron mass exponent (Doc.~320, Eq.~5),
    the fundamental sub-Planck suppression of the first lepton
    generation.
  \item $1/(4N_c) = 1/12$: discrete phase correction of the
    four-dimensional $\mathbb{Z}_3$ fibration on $T^4$.
    The factor $4N_c = 12$ counts the product of torus dimensions
    (4) and colour degrees of freedom ($N_c=3$, Doc.~321).
\end{itemize}

\textbf{[K]} Numerically:
\begin{equation}
  M_{\rm GUT} = 1.221\times10^{19}\,\text{GeV}\cdot\xi^{19/12}
  = 8.94\times10^{12}\,\text{GeV},
\end{equation}
in the range $10^{12}$--$10^{15}$~GeV typical for non-supersymmetric
GUT models.

% =========================================================
\section{Step 3: $\sin^2\theta_W(M_Z)$ from the GQW Formula}
% =========================================================

\textbf{[B]} The 1-loop $SU(5)$ result (Georgi, Quinn, Weinberg 1974):
\begin{equation}
  \sin^2\theta_W(\mu)
  = \frac{3}{8}
  - \frac{55\,\alpha_{\rm em}(\mu)}{24\pi}
    \ln\!\frac{M_{\rm GUT}}{\mu}.
\end{equation}

With $M_{\rm GUT} = m_{\rm Pl}\cdot\xi^{19/12}$:
\begin{align}
  \ln\!\frac{M_{\rm GUT}}{M_Z}
  &= \ln\!\frac{m_{\rm Pl}}{M_Z} + \frac{19}{12}\ln\xi \nonumber\\
  &= 39.44 + \tfrac{19}{12}\times(-8.923)
   = 39.44 - 14.13 = 25.31.
\end{align}

\textbf{[K]} Result:
\begin{align}
  \sin^2\theta_W(M_Z)
  &= \frac{3}{8}
   - \frac{55}{24\pi}\cdot\frac{1}{128}\cdot25.31 \nonumber\\
  &= 0.3750 - 0.1442
   = \mathbf{0.2308}.
\end{align}

\begin{equation}
  \boxed{\sin^2\theta_W(M_Z)_{\rm FFGFT}
  = \frac{3}{8}
  - \frac{55\,\alpha_{\rm em}(M_Z)}{24\pi}
    \Bigl[\ln\!\frac{m_{\rm Pl}}{M_Z}
         + \frac{19}{12}\ln\xi\Bigr]
  = 0.2308}
\end{equation}

PDG: $0.2312$. Deviation: $-0.19\,\%$.
