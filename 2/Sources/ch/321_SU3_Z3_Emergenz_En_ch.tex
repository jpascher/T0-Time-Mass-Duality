\section{Starting Point: $\mathbb{Z}_3$ as the Centre of $SU(3)$}
% =========================================================

\subsection{The centre of a Lie group}

The centre $Z(G)$ of a group $G$ consists of all elements that
commute with every other element: $Z(G) = \{z \in G : zg = gz
\;\forall g\in G\}$.

\textbf{[K]} For $SU(3)$:
\begin{equation}
  Z(SU(3)) = \mathbb{Z}_3 =
  \left\{\mathbf{1},\;\omega\mathbf{1},\;\omega^2\mathbf{1}\right\},
  \quad \omega = e^{2\pi i/3}.
  \label{eq:centre}
\end{equation}

\textbf{[K]} This is not a coincidence: $\mathbb{Z}_3 = Z(SU(3))$
means that any $\mathbb{Z}_3$ symmetry embeddable into the unitary
group leads automatically into the $SU(3)$ structure. The $\mathbb{Z}_3$
action on $T^4/\mathbb{Z}_3$ is therefore not an additional FFGFT
assumption but the minimal unitary extension of the $\mathbb{Z}_3$
centre to a full gauge group.

\subsection{Triality: the $\mathbb{Z}_3$ charge of $SU(3)$ representations}

\textbf{[B]} Every irreducible representation of $SU(3)$ has a
\emph{triality} $T_R \in \{0,1,2\}$, defined by the
$\mathbb{Z}_3 = Z(SU(3))$ action:
\begin{equation}
  \omega\mathbf{1} \xrightarrow{\text{repr.}\;R}
  \omega^{T_R}\cdot\mathbf{1}_R.
\end{equation}

\section{The $\mathbb{Z}_3$ Sector Decomposition on $L^2(T^4)$}
% =========================================================

\subsection{Three projectors from one $\mathbb{Z}_3$ action}

\textbf{[K]} The $\mathbb{Z}_3$ group $\{e,g,g^2\}$ acts on $T^4$
through the unitary operator $\tau$ ($\tau^3 = \mathrm{id}$). On
$L^2(T^4)$ this induces three orthogonal projectors:
\begin{equation}
  P_k = \frac{1}{3}\sum_{j=0}^{2}\omega^{-jk}\tau^j,
  \quad k=0,1,2.
  \label{eq:projectors}
\end{equation}

\textbf{[B]} These projectors are pairwise orthogonal and complete:
\begin{equation}
  P_k^2 = P_k,\quad P_kP_l = \delta_{kl}P_k,\quad
  P_0+P_1+P_2 = \mathrm{id}.
\end{equation}

\emph{Proof of idempotency:}
$P_k^2 = \frac{1}{9}\sum_{j,j'}\omega^{-k(j+j')}\tau^{j+j'}
= \frac{1}{9}\sum_n 3\,\omega^{-kn}\tau^n = P_k$. \checkmark

\subsection{The three eigensectors and their modes}

\textbf{[K]} In the Fourier basis $e^{in\cdot x/R}$ the
$\mathbb{Z}_3$ action $(x_1,x_2,x_3,x_4)\mapsto(x_2,x_3,x_1,x_4)$
gives sector index $k \equiv n_1+n_2+n_3 \pmod{3}$:
\begin{itemize}
  \item $\mathcal{H}_0$: $n_1+n_2+n_3\equiv0$ — colour-neutral
    (gluons, singlets).
  \item $\mathcal{H}_1$: $n_1+n_2+n_3\equiv1$ — quarks.
  \item $\mathcal{H}_2$: $n_1+n_2+n_3\equiv2$ — antiquarks.
\end{itemize}

\section{Algebraic Construction of the Eight Gell-Mann Operators}
% =========================================================

\subsection{Bilinear transitions between sectors}

\textbf{[B]} The diagonal basis $(e_0,e_1,e_2)$ with
$\tau e_k = \omega^k e_k$ corresponds to the three colour states
(Red $=e_0$, Green $=e_1$, Blue $=e_2$). The $3^2-1=8$ traceless
Hermitian operators on $\mathbb{C}^3$ are constructed as:

\medskip
\textit{Transitions $\mathcal{H}_0\leftrightarrow\mathcal{H}_1$:}
$\quad T_1 = \tfrac{1}{2}(|e_0\rangle\langle e_1|+|e_1\rangle\langle e_0|)$,
$\quad T_2 = \tfrac{i}{2}(|e_1\rangle\langle e_0|-|e_0\rangle\langle e_1|)$.

\textit{Transitions $\mathcal{H}_0\leftrightarrow\mathcal{H}_2$:}
$\quad T_4 = \tfrac{1}{2}(|e_0\rangle\langle e_2|+|e_2\rangle\langle e_0|)$,
$\quad T_5 = \tfrac{i}{2}(|e_2\rangle\langle e_0|-|e_0\rangle\langle e_2|)$.

\textit{Transitions $\mathcal{H}_1\leftrightarrow\mathcal{H}_2$:}
$\quad T_6 = \tfrac{1}{2}(|e_1\rangle\langle e_2|+|e_2\rangle\langle e_1|)$,
$\quad T_7 = \tfrac{i}{2}(|e_2\rangle\langle e_1|-|e_1\rangle\langle e_2|)$.

\textit{Diagonal operators:}
$\quad T_3 = \tfrac{1}{2}(|e_0\rangle\langle e_0|-|e_1\rangle\langle e_1|)$,
$\quad T_8 = \tfrac{1}{2\sqrt{3}}(|e_0\rangle\langle e_0|+|e_1\rangle\langle e_1|-2|e_2\rangle\langle e_2|)$.

  \label{eq:gellmann}

\section{Proof of the $\mathfrak{su}(3)$ Commutation Relations}
% =========================================================

\subsection{Normalisation condition}

\textbf{[B]} Probe for $a=b=1$:
\begin{equation}
  \mathrm{tr}(T_1T_1) = \tfrac{1}{4}\mathrm{tr}(\lambda_1^2)
  = \tfrac{1}{4}\mathrm{tr}
  \begin{pmatrix}1&0&0\\0&1&0\\0&0&0\end{pmatrix}
  = \tfrac{2}{4} = \tfrac{1}{2}.\quad\checkmark
\end{equation}

\subsection{Selected commutators}

\textbf{[B]}

$[T_1,T_2] = iT_3$:
\begin{equation}
  [T_1,T_2] = \tfrac{1}{4}[\lambda_1,\lambda_2]
  = \tfrac{i}{2}
  \begin{pmatrix}1&0&0\\0&{-1}&0\\0&0&0\end{pmatrix}
  = iT_3.\quad\checkmark
\end{equation}

$[T_1,T_4] = \tfrac{i}{2}T_7$: $T_1$ connects
$\mathcal{H}_0\leftrightarrow\mathcal{H}_1$, $T_4$ connects
$\mathcal{H}_0\leftrightarrow\mathcal{H}_2$, so their commutator
falls in the $\mathcal{H}_1\leftrightarrow\mathcal{H}_2$ sector.
The imaginary transition $T_7$ appears (not $T_6$):
\begin{equation}
  [T_1,T_4] = \tfrac{1}{4}[\lambda_1,\lambda_4]
  = \tfrac{i}{2}T_7.\quad\checkmark
\end{equation}
Structure constant: $f_{147} = 1/2$ (canonical $\mathfrak{su}(3)$ value).

$[T_3,T_8] = 0$: both diagonal; diagonal matrices commute. \checkmark

$T_3$ and $T_8$ generate the Cartan subalgebra $\mathfrak{h}
\subset\mathfrak{su}(3)$ (maximum abelian subalgebra, dimension~2).

\begin{keyresult}[Main result Doc.~321 \textbf{[B]}]
The eight bilinear transition operators between the three
$\mathbb{Z}_3$ eigensectors of $L^2(T^4)$ satisfy the
$\mathfrak{su}(3)$ Lie algebra relations $[T_a,T_b]=if_{abc}T_c$

The $SU(3)_c$ gauge group emerges algebraically necessarily from the

\end{keyresult}

\begin{equation}

  = \underbrace{T^3/\mathbb{Z}_3}_{SU(3)_c}

\end{equation}

\subsection{$U(1)_Y$ from the fourth torus direction}

\textbf{[B]} The fourth torus direction carries a global $U(1)$ phase

$\hat{Y} = \partial/\partial\alpha|_{\alpha=0}$.
It acts as $\hat{Y}|n_4\rangle = n_4|n_4\rangle$, quantising
hypercharge by the fourth winding number.
Compatibility with $\mathbb{Z}_3$ forces $Y\in\tfrac{1}{3}\mathbb{Z}$.

\subsection{$SU(2)_L$ from the $\mathbb{Z}_2$ pairing of sectors}

\textbf{[B]} Complex conjugation $\mathcal{C}:\mathcal{H}_1

$\sigma_1,\sigma_2,\sigma_3$ with
$[\sigma_i/2,\sigma_j/2]=i\varepsilon_{ijk}\sigma_k/2$
