\section*{Abstract}

Dok.~358 und 359 liefern die algebraische Begründung für einen
Galois-informierten Ansatz in der HRV-Frequenzanalyse. Dieses
Dokument implementiert den Ansatz als Python-Modul und validiert
ihn an zwei öffentlichen EKG-Datensätzen. Das Ergebnis ist eine
\emph{Negativ-Validierung} mit klarem strukturellen Grund.

\textbf{Ergebnisse:}
\begin{enumerate}
\item \textbf{Modul} \KM: \texttt{galois\_hrv\_preprocess.py} —
  vollständige Pipeline: Galois-Projektion, Auflösungsfilter
  $100\xi\approx1{,}33\,\%$, HRV-Bandgrenzen als Konstanten,
  Orbit-Typ-Klassifikation, Arnold-Zungenbreite.
\item \textbf{BIDMC-01} \LM: ICU-Patient, 88~J., 8~min PPG-EKG,
  Pimentel et al.\ 2017. Ergebnis: 0/3 Verhältnisse 5-Limit,
  LF/VLF nicht auflösbar (zu kurz). Struktureller Grund: Aufnahme
  zu kurz für VLF, DFA $\alpha\approx0{,}5$ (weißes Rauschen,
  fraktale Regulation verloren).
\item \textbf{Fantasia} \LM: 5 junge + 5 alte Gesunde, je $\approx120$~min
  Ruhe, spontane Atmung, Gold et al.\ 2002. Ergebnis: LF-Peak-CV
  11--33\,\% $\gg$ 1{,}33\,\%, 5-Limit LF/HF 4/10 (40\,\%),
  kein Gruppenunterschied jung/alt.
\item \textbf{Struktureller Grund} \BM: Bei spontaner Atmung ist
  $\text{HF} = \text{Atemfrequenz}$ (variabel). LF/HF ist das
  Verhältnis zweier unabhängiger Regelkreise und kein festes
  Galois-Verhältnis.
\item \textbf{Nächster Schritt} \SM: Kontrollierte Atmung 6/min
  (Resonanzfrequenz-Protokoll, Lehrer \& Gevirtz 2014) mit
  Polar H10 Brustgurt.
\end{enumerate}

\section{Galois-HRV-Preprocessing-Modul}
% ============================================================

\texttt{galois\_hrv\_preprocess.py} implementiert (Dok.~359 §5):

\begin{description}
\item[Galois-Raster] 901 rationale Verhältnisse $p/q$ mit Primfaktoren
  $\subseteq\{2,3,5,7,11,13\}$ bis Nenner 60.
\item[Auflösungsfilter] $\Delta f/f \geq 100\xi \approx 1{,}33\,\%$
  (Farey-$Q_c\approx11$, Dok.~343).
\item[HRV-Bandgrenzen] VLF: $[0,\,\tfrac{1}{25}]$~Hz, LF: $[\tfrac{1}{25},\,\tfrac{3}{20}]$~Hz,
  HF: $[\tfrac{3}{20},\,\tfrac{2}{5}]$~Hz --- 5-Limit-Konstanten \BM\ (Dok.~359 Satz~A).
\item[Galois-Projektion] LF/HF $\mapsto$ nächstes $p/q$ im Raster.
\item[Orbit-Typ] Galois-Tiefe $k=\mathrm{ord}_p(3)$ der Primfaktoren
  (Dok.~343, 358).
\item[Stabilität] Arnold-Zungenbreite $\approx 2/(p+q)$ (Dok.~328).
\item[Pipeline] \texttt{hrv\_galois\_pipeline(rr\_ms)} --- ein Funktionsaufruf.
\end{description}

\section{Negativ-Validierung I: BIDMC-01}
% ============================================================

\textbf{Datensatz:} BIDMC PPG and Respiration Dataset v1.0.0,
Record 01 \LM\,\cite{Pimentel2017}.
88~J., m., MICU, 8~min, 125~Hz PPG+Respiration.

\textbf{Ergebnis:} 0/3 Frequenzverhältnisse erfüllen das 5-Limit-Kriterium
(Welch und Lomb-Scargle). VLF nicht auflösbar (Aufnahmedauer 8~min $<$ 15~min
Minimum für VLF). DFA $\alpha_1=0{,}50$, $\alpha_2=0{,}45$ --- weißes Rauschen;
bei Gesunden $\approx1{,}0$.

\textbf{Warum:}
\begin{enumerate}
\item Zu kurz für VLF ($<15$~min).
\item ICU-Patient: fraktale Regulation weitgehend verloren.
\item HF-Peak = gemessene Atemfrequenz (0{,}328~Hz = RESP-Kanal 0{,}336~Hz).
\item Messunsicherheit Welch vs.\ Lomb-Scargle: $\pm2\,\%$ $>$ FFGFT-Grenze
  1{,}33\,\%.
\end{enumerate}

\end{itemize}

\section{Struktureller Grund und Konsequenz}
% ============================================================

\begin{quote}
\textbf{Befund.} \BM\ Bei spontaner Atmung ist HF = Atemfrequenz
(variabel 0{,}15--0{,}40~Hz). LF/HF ist das Verhältnis zweier
unabhängiger physiologischer Regelkreise (Barorezeptor~/ Atmung)
und konvergiert nicht gegen ein festes Galois-Verhältnis.
Der Galois-Test ist bei spontaner Atmung strukturell nicht prüfbar.
\end{quote}

\textbf{Nächster Schritt \SM:} Kontrollierte Atmung bei 6~Zügen/min
(0{,}10~Hz = LF-Resonanzfrequenz). Dann fallen HF und LF zusammen;
das System kann einrasten (Arnold-Zunge breit) oder nicht.
Das ist der Lehrer-\&-Gevirtz-Effekt \LM\,\cite{Lehrer2014},
physiologisch gut belegt.

\textbf{Empfohlenes Setup:}
\begin{itemize}
\item Polar H10 Brustgurt (1~ms RR-Auflösung, $\approx90$~€).
\item 5~min Eingewöhnung, 10~min bei 6/min Metronom.
\item Export via Elite HRV oder Kubios HRV (CSV).
\item Analyse mit \texttt{galois\_hrv\_preprocess.py}.
\end{itemize}
