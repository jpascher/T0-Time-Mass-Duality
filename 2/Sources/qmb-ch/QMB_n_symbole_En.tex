\section*{Basic Quantities of the FFGFT}

\begin{description}[leftmargin=4.5em,style=sameline,itemsep=0.4em]
  \item[$\xi$] Geometry parameter of the FFGFT, $\xi = 4/30000 =
    1/7500 \approx 1{,}333\times10^{-4}$. The only free parameter;
    follows from Galois group orders of the carrier.
  \item[$\tilde T$] Compact time coordinate of a mode; $\tilde T =
    1/m$ in natural units ($\hbar = c = 1$).
  \item[$T^4/\mathbb{Z}_3$] FFGFT carrier: four-dimensional torus with
    $\mathbb{Z}_3$ symmetry. Three spatial and one temporal
    winding direction; nine fixed points, triality symmetry.
  \item[$K_{\mathrm{frak}}$] Fractal correction factor,
    $K_{\mathrm{frak}} = 1 - 100\xi \approx 0{,}9867$. Arises during the
    Type I transition from the compact carrier to the SI scale; cumulative over
    17 orders of magnitude (RG flow from Planck to electroweak scale).
  \item[$D_f$] Fractal spatial dimension, $D_f = 3 - \xi \approx
    2{,}9999$. Local; not to be confused with $K_{\mathrm{frak}}$.
  \item[$L_0$] Fundamental FFGFT length, $L_0 = \xi\,\ell_P \approx
    2{,}15\times10^{-39}$\,m.
  \item[$\ell_P$] Planck length, $\ell_P \approx 1{,}616\times10^{-35}$\,m.
  \item[$t_P$] Planck time, $t_P \approx 5{,}39\times10^{-44}$\,s.
\end{description}

\section*{Qubit and Quantum Computer Symbols}

\begin{description}[leftmargin=4.5em,style=sameline,itemsep=0.4em]
  \item[$(z,r,\theta)$] Cylindrical phase space coordinates of an
    FFGFT qubit: $z\in[-1,+1]$ projection onto the computation axis,
    $r\in[0,1]$ radial superposition amplitude, $\theta\in[0,2\pi)$
    relative phase.
  \item[$(\alpha,\beta)$] Standard QM amplitudes of a qubit:
    $|\psi\rangle = \alpha|0\rangle+\beta|1\rangle$,
    $|\alpha|^2+|\beta|^2 = 1$.
  \item[$K_{\mathrm{frak}}$ (gate)] In the $\pi$ gates
    ($\sigma_x$ rotation): rotation angle $\alpha = \pi K_{\mathrm{frak}}$
    instead of $\pi$. Derived geometrically from $D_f$; testable prediction.
  \item[$\hat F$] Fundamental fractal spectral operator on
    $\mathcal{H} = L^2(T^4/\mathbb{Z}_3)\otimes\mathbb{C}^3$.
    Self-adjoint; $\xi = \lambda_{\min}(\hat F_{D_4})$.
  \item[$D_4$] Root lattice in $\mathbb{R}^4$: all integer vectors with even coordinate sum. Kissing number 24; basis of the
    mode spectrum.
  \item[$\mathrm{CHSH}$] Clauser-Horne-Shimony-Holt parameter;
    local-realistic $\le2$, Tsirelson bound (QM)
    $2\sqrt2\approx2{,}828$.
  \item[$\Delta\mathrm{CHSH}_{\mathrm{elem}}$] Elementary
    FFGFT deviation per measurement: $\xi/(2\pi)\approx2\times10^{-5}$.
    Hypothesis; formal derivation pending.
  \item[$T_1,\,T_2$] Relaxation time ($T_1$) and coherence time ($T_2$)
    of a qubit; always $T_2\le2T_1$.
  \item[$Q,\,Q_\varphi$] Register dimension: $Q = 2^n$ (standard QFT),
    $Q_\varphi = \varphi^n$ ($\phi$-QFT). Both equivalent for
    period finding up to $N < 2^{20}$ on $\pm\xi$.
\end{description}

\section*{Spectral Theory and Lattices}

\begin{description}[leftmargin=4.5em,style=sameline,itemsep=0.4em]
  \item[$N(n)$] Degeneracy of the shell $|k|^2 = n$ in the $D_4$ lattice;
    coefficient of the theta series
    $\theta_{D_4} = (\theta_3^4+\theta_4^4)/2$.
  \item[$Z_{\mathbb{Z}^4}(s)$] Spectral zeta of the cubic torus:
    $8(1-4^{1-s})\zeta(s)\zeta(s-1)$.
  \item[$\mathrm{Aut}(D_4)$] Automorphism group of the $D_4$ lattice;
    $|\mathrm{Aut}(D_4)| = 1152 = 2^7\cdot3^2$. Coprime to 5:
    no fivefold symmetry.
  \item[$\chi_0,\chi_1,\chi_2$] The three $\mathbb{Z}_3$ characters of the
    orbifold; each $D_4$ shell decomposes uniformly into
    $N(n)/3$ states per sector.
  \item[$\lambda_4$] Arc length on the mass circle; de Broglie relation
    $\lambda_4\cdot m = 2\pi$ exact per mode (position relation, not
    time relation).
\end{description}

\section*{Time Structure}

\begin{description}[leftmargin=4.5em,style=sameline,itemsep=0.4em]
  \item[Type I] Measurement act (unrolling): lossy, $S^1_m\to\mathbb{R}_t$.
    Information loss, time arrow geometric.
  \item[Type III] Representation change (pullback): lossless, bijective.
    No information loss; $K_{\mathrm{frak}}$ not involved.
  \item[$\xi_{\mathrm{Higgs}}$] Higgs coupling parameter,
    $\approx 1{,}038\times10^{-5}$. Acts in the algebraic
    state space (coupling rates, decoherence); \emph{not} identical
    with $\xi$, which acts in geometric position space.
\end{description}

\section*{Status Markers}

\begin{description}[leftmargin=2.5em,style=sameline,itemsep=0.3em]
  \item[\textbf{[K]}] Confirmed — fully proven from $\xi$ or a
    corpus-secured derivation; test script passed.
  \item[\textbf{[B]}] Documented — numerically secured; analytical proof
    present or trivial.
  \item[\textbf{[S]}] Speculative — hypothesis or indication; not yet
    fully derived.
  \item[\textbf{[E]}] Established physics — standard result, here only
    recalled.
  \item[\textbf{[X]}] Checked negative — explicitly excluded with
    justification.
  \item[\textbf{[SETZUNG]}] Axiomatic postulate — starting point of the
    derivation, not further justified.
\end{description}
