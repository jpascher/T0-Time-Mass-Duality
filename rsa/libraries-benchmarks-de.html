🔬 Technische Details

# Rationale Arithmetik in der Ordnungssuche, Gleitkomma in der harmonischen Bibliothek import math from fractions import Fraction # Exakte Verhältnisse statt Fließkommazahlen pi_exact = Fraction(355, 113) # π-Näherung sigma = Fraction(1, 10) # Rasterparameter σ (ohne Einfluss auf das Ergebnis) # Ordnungssuche: Periodenbewertung mit exakten Verhältnissen def calculate_period_score(r, sigma): omega = Fraction(2) * pi_exact / Fraction(r) diff = omega - pi_exact score = Fraction(1) / (Fraction(1) + diff * diff / (4 * sigma)) return score # Harmonisch: Logarithmische Cents-Berechnung def calculate_cents_deviation(ratio1, ratio2): return abs(1200 * math.log2(float(ratio1 / ratio2)))

Warum rationale Arithmetik?

  • Bitgenau reproduzierbare Bewertung auf jeder Hardware (R65)
  • Keine Rundung innerhalb der Bruchrechnung; π selbst ist durch 355/113 angenähert
  • Auf die gefundenen Faktoren hat das keinen Einfluss: die Ordnungssuche rechnet ganzzahlig

🎯 Praktische Anwendung

# Beispiel: Vollständiger Ablauf mit beiden Ansätzen from t0_rational_optimized import T0RationalSimulatorOptimized from optimized_harmonic_lib import OptimizedHarmonicFactorizer def analyze_number(n): """Analysiere Zahl mit beiden Ansätzen""" # Ordnungssuche t0 = T0RationalSimulatorOptimized(n) t0_result = {'factors': t0.shor_t0_rational() or None} # Harmonische Klassifikation (Probedivision, dann Benennung) harmonic = OptimizedHarmonicFactorizer() harmonic_result = harmonic.factorize(n) print(f"Zahl: {n}") print(f"Ordnungssuche erfolgreich: {t0_result['factors'] is not None}") print(f"Harmonische Zuordnung: {harmonic_result.success}") if harmonic_result.success: print(f"Musikalisches Intervall: {harmonic_result.harmonic_name}") print(f"Hierarchie-Level: {harmonic_result.level_name}") return t0_result, harmonic_result # Test verschiedener Zahlentypen test_numbers = [221, 323, 1643, 2491, 10403] for num in test_numbers: analyze_number(num) # Ergebnis: Die Ordnungssuche zerlegt 221, 323 und 1643; für 2491 und 10403 # liefert sie keine Faktoren. Die harmonische Bibliothek benennt alle fünf.
